              jumpToStop(peekStop.id);
            }}
          />
        )}
        {/* Past You, for a trip still ahead or under way: not one already over. */}
        {perspective === "overview" &&
          todayKey <= (trip.end_date ?? trip.start_date ?? "9999-12-31") && (
            <div className="mb-3">
              <PastYouCard
                trip={trip}

            </div>
          )}
